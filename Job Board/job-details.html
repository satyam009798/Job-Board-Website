const search=document.getElementById("jobSearch"); if(search){ search.addEventListener("keyup",function(){ let filter=search.value.toLowerCase(); let cards=document.querySelectorAll(".job-card"); cards.forEach(function(card){ let title=card.querySelector("h3").innerText.toLowerCase(); if(title.includes(filter)){ card.style.display="block"; } else{ card.style.display="none"; } }); }); } const buttons=document.querySelectorAll(".apply-btn"); buttons.forEach(function(btn){ btn.onclick=function(){ window.location.href="apply.html"; }; });
